// Integers have 4 bytes
// Each byte has 256 possible values
bucket = [4][256]

// You begin by placing each item into the buckets
for integer in input
    bump bucket[1] count of integer (byte 4) value by 1
    bump bucket[2] count of integer (byte 3) value by 1
    bump bucket[3] count of integer (byte 2) value by 1
    bump bucket[4] count of integer (byte 1) value by 1 (ignore sign bit)
// The above is 4 operations per item in the input, or O(4n)

// Then you do a prefix sum
// this tells you (depending on the number of other elements)

    bucket byte 2 (sum all previous)
    bucket byte 3 (sum all previous)
    bucket byte 4 (sum all previous)
